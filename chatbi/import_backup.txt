# -*- coding: utf-8 -*-
import openpyxl
import pymysql

EXCEL_PATH = r'D:\OpenCode-Project\aiagent\chatbi\final_merged_stock_data_20200101_to_20251220.xlsx'
SQL_PATH = r'D:\OpenCode-Project\aiagent\chatbi\stock_data.sql'

wb = openpyxl.load_workbook(EXCEL_PATH, read_only=True)
ws = wb.active
rows = ws.iter_rows(values_only=True)
headers = list(next(rows))
data_rows = list(rows)
print('Headers:', headers)
print('Data rows:', len(data_rows))

cols_names = ', '.join('' + h + '' for h in headers)
col_placeholder = ', '.join(['%s'] * len(headers))
col_defs = {
    'ts_code': 'VARCHAR(20)', 'trade_date': 'VARCHAR(12)',
    'open': 'DECIMAL(12,4)', 'high': 'DECIMAL(12,4)', 'low': 'DECIMAL(12,4)',
    'close': 'DECIMAL(12,4)', 'pre_close': 'DECIMAL(12,4)', 'change': 'DECIMAL(12,4)',
    'pct_chg': 'DECIMAL(10,4)', 'vol': 'DECIMAL(18,4)', 'amount': 'DECIMAL(18,4)',
    'stock_name': 'VARCHAR(50)'
}

lines = []
lines.append('-- Stock data from Tushare')
lines.append('CREATE DATABASE IF NOT EXISTS stock_data DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;')
lines.append('USE stock_data;')
lines.append('')
lines.append('DROP TABLE IF EXISTS daily_kline;')
col_create = ',\n  '.join('' + h + ' ' + col_defs.get(h, 'VARCHAR(255)') for h in headers)
lines.append('CREATE TABLE daily_kline (')
lines.append('  id BIGINT AUTO_INCREMENT PRIMARY KEY,')
lines.append('  ' + col_create + ',')
lines.append('  UNIQUE KEY uk_ts_trade (	s_code, 	rade_date),')
lines.append('  INDEX idx_trade_date (	rade_date),')
lines.append('  INDEX idx_ts_code (	s_code),')
lines.append('  INDEX idx_stock_name (stock_name)')
lines.append(') ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;')
lines.append('')

def sql_escape(v):
    if v is None:
        return 'NULL'
    elif isinstance(v, str):
        return "'" + v.replace("'", "\\'") + "'"
    elif isinstance(v, (int, float)):
        return str(v)
    else:
        return "'" + str(v).replace("'", "\\'") + "'"

batch_size = 500
for i in range(0, len(data_rows), batch_size):
    batch = data_rows[i:i+batch_size]
    vals = []
    for r in batch:
        esc = [sql_escape(v) for v in r]
        vals.append('(' + ', '.join(esc) + ')')
    lines.append('INSERT INTO daily_kline (' + cols_names + ') VALUES ')
    lines.append(','.join(vals) + ';')
    lines.append('')

sql = '\n'.join(lines)
with open(SQL_PATH, 'w', encoding='utf-8') as f:
    f.write(sql)
print('SQL file written:', SQL_PATH)

# Insert into DB
conn = pymysql.connect(host='127.0.0.1', port=3306, user='root', password='root', charset='utf8mb4')
cur = conn.cursor()
cur.execute('CREATE DATABASE IF NOT EXISTS stock_data DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci')
cur.execute('USE stock_data')
cur.execute('DROP TABLE IF EXISTS daily_kline')
cur.execute(
    'CREATE TABLE daily_kline ('
    '  id BIGINT AUTO_INCREMENT PRIMARY KEY,'
    '  	s_code VARCHAR(20), 	rade_date VARCHAR(12),'
    '  open DECIMAL(12,4), high DECIMAL(12,4), low DECIMAL(12,4), close DECIMAL(12,4),'
    '  pre_close DECIMAL(12,4), change DECIMAL(12,4), pct_chg DECIMAL(10,4),'
    '  ol DECIMAL(18,4), mount DECIMAL(18,4), stock_name VARCHAR(50),'
    '  UNIQUE KEY uk_ts_trade (	s_code, 	rade_date),'
    '  INDEX idx_trade_date (	rade_date),'
    '  INDEX idx_ts_code (	s_code),'
    '  INDEX idx_stock_name (stock_name)'
    ') ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci'
)
insert_sql = 'INSERT INTO daily_kline (' + cols_names + ') VALUES (' + col_placeholder + ')'
cur.executemany(insert_sql, data_rows)
conn.commit()
cur.execute('SELECT COUNT(*) FROM daily_kline')
total = cur.fetchone()[0]
print('Done! Total records in daily_kline:', total)
cur.close()
conn.close()
print('Database import complete.')
